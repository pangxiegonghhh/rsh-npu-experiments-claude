cmake_minimum_required(VERSION 3.16)
project(e1_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN")
set(SOC_VERSION Ascend910B2 CACHE STRING "SoC")
set(ASCEND_COMPUTE_UNIT ${SOC_VERSION})
set(RUN_MODE npu)
set(ASCENDC_CMAKE "${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
if(NOT EXISTS "${ASCENDC_CMAKE}")
 set(ASCENDC_CMAKE "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
endif()
include(${ASCENDC_CMAKE})
ascendc_library(e1_decode_lib STATIC decode.cpp)
ascendc_library(e1_cube_lib STATIC cube.cpp)
add_executable(e1_runner host.cpp)
target_include_directories(e1_runner PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(e1_runner PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(e1_runner PRIVATE e1_decode_lib e1_cube_lib ascendcl)

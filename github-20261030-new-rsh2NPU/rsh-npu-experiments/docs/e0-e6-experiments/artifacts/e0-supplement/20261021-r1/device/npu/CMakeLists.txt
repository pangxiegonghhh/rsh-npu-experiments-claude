cmake_minimum_required(VERSION 3.16)
project(rsh_910b LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "Actual CANN toolkit")
set(SOC_VERSION "Ascend910B2" CACHE STRING "Verified target SoC")
set(ASCEND_COMPUTE_UNIT ${SOC_VERSION})
set(RUN_MODE npu)
set(ASCENDC_CMAKE "${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
if(NOT EXISTS "${ASCENDC_CMAKE}")
 set(ASCENDC_CMAKE "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
endif()
if(NOT EXISTS "${ASCENDC_CMAKE}")
 message(FATAL_ERROR "Cannot find Ascend C toolchain under ASCEND_CANN_PACKAGE_PATH")
endif()
include(${ASCENDC_CMAKE})
ascendc_library(rsh_kernels STATIC cube.cpp cube_direct.cpp vector.cpp mix.cpp restore.cpp)
add_executable(rsh_runner runner.cpp dispatch.cpp)
target_include_directories(rsh_runner PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(rsh_runner PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(rsh_runner PRIVATE rsh_kernels ascendcl)

# E0-only capture/guard runner, production runner remains unchanged.
add_executable(rsh_guarded runner_guarded.cpp dispatch.cpp)
target_include_directories(rsh_guarded PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(rsh_guarded PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(rsh_guarded PRIVATE rsh_kernels ascendcl)

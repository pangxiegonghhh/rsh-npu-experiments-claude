cmake_minimum_required(VERSION 3.16)
project(e0_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN")
set(SOC_VERSION Ascend910B2 CACHE STRING "SoC")
set(ASCEND_COMPUTE_UNIT ${SOC_VERSION})
set(RUN_MODE npu)
set(ASCENDC_CMAKE "${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
if(NOT EXISTS "${ASCENDC_CMAKE}")
 set(ASCENDC_CMAKE "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
endif()
include(${ASCENDC_CMAKE})
foreach(mode normal direct dual)
 ascendc_library(e0_kernel_${mode} STATIC ${mode}.cpp)
 add_executable(e0_${mode} host.cpp)
 if(mode STREQUAL "dual")
  target_compile_definitions(e0_${mode} PRIVATE E0_DUAL)
 endif()
 target_include_directories(e0_${mode} PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
 target_link_directories(e0_${mode} PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
 target_link_libraries(e0_${mode} PRIVATE e0_kernel_${mode} ascendcl)
endforeach()

ascendc_library(e0_kernel_split STATIC normal.cpp direct_dual.cpp)
add_executable(e0_split host.cpp)
target_compile_definitions(e0_split PRIVATE E0_DUAL)
target_include_directories(e0_split PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(e0_split PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(e0_split PRIVATE e0_kernel_split ascendcl)

cmake_minimum_required(VERSION 3.16)
project(t8_micro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}")
set(SOC_VERSION Ascend910B2)
set(ASCEND_COMPUTE_UNIT ${SOC_VERSION})
set(RUN_MODE npu)
include(${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(t8_kernel STATIC kernel.cpp)
add_executable(t8_micro host.cpp)
target_include_directories(t8_micro PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(t8_micro PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(t8_micro PRIVATE t8_kernel ascendcl)

add_executable(t8_identity identity_host.cpp)
target_include_directories(t8_identity PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(t8_identity PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(t8_identity PRIVATE t8_kernel ascendcl)

ascendc_library(b3_kernel STATIC kernel-b3.cpp)
add_executable(m11b_b3_gate b3_host.cpp)
target_include_directories(m11b_b3_gate PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(m11b_b3_gate PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(m11b_b3_gate PRIVATE b3_kernel ascendcl)

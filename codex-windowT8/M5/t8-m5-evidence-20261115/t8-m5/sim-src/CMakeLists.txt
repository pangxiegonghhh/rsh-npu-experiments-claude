cmake_minimum_required(VERSION 3.16)
project(t8_micro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}")
set(SOC_VERSION Ascend910B2)
set(ASCEND_COMPUTE_UNIT ${SOC_VERSION})
set(RUN_MODE sim)
include(${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(t8_kernel STATIC kernel.cpp)
ascendc_compile_definitions(t8_kernel PRIVATE ASCENDC_TRACE_ON)
ascendc_compile_options(t8_kernel PRIVATE -g -O3)
add_executable(t8_micro host.cpp)
target_include_directories(t8_micro PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(t8_micro PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(t8_micro PRIVATE t8_kernel ascendcl)
target_link_directories(t8_micro PRIVATE /usr/local/Ascend/cann-9.0.0/tools/simulator/Ascend910B2/lib)
target_link_libraries(t8_micro PRIVATE -Wl,--no-as-needed model_top -Wl,--as-needed)

target_link_libraries(t8_micro PRIVATE -Wl,--no-as-needed mcu_loop mcu_wrapper pem_davinci stars stars_pv -Wl,--as-needed)

target_link_libraries(t8_micro PRIVATE -Wl,--no-as-needed ffts_model -Wl,--as-needed)
